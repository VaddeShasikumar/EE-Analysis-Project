   "metadata": {},
   "execution_count": null,
   "outputs": [],
   "source": [
    "profile.columns = profile.columns.str.lower().str.replace('foriegn', 'foreign')\n",
    "profile['occupation_level'] = profile['occupation_level'].replace({'Temporary': 'Temporary employees'})\n",
    "\n",
    "print(profile['occupation_level'].value_counts())            # 7 levels x 6000 rows\n",
    "print('duplicate keys:', profile.duplicated(['submission_year','ee_ref','occupation_level']).sum())\n",

  {
   "cell_type": "markdown",

   "source": [
    "## 2. Reshape the meta file (one row per employer-year)\n",
    "The meta file has one row per employer and one column per attribute per year (73 columns)."
   ]
  },
  {
   "cell_type": "code",
   "metadata": {},
   "execution_count": null,
   "outputs": [],
   "source": [
    "stubs = ['dti_registration_name','industry_sector','seta_classification',\n",
    "         'business_type','province_physical','city_town_physical']\n",
    "meta_long = pd.wide_to_long(meta, stubnames=stubs, i='ee_ref',\n",
    "                            j='submission_year', sep='_').reset_index()\n",
    "for c in stubs:\n",
    "    meta_long[c] = meta_long[c].str.strip().str.upper()\n",
    "print(meta_long.shape)   # 6000 = 500 x 12"
   ]
  },
  {
   "cell_type": "markdown",
   "metadata": {},
   "source": [
    "## 3. Merge"
   ]
  },
  {
   "cell_type": "code",
   "metadata": {},
   "execution_count": null,
   "outputs": [],
   "source": [
    "df = profile.merge(meta_long, on=['ee_ref','submission_year'], how='left', validate='m:1')\n",
    "assert len(df) == len(profile) == 42000\n",
    "race_gender = [c for c in df.columns if c.endswith(('_male','_female'))]\n",
    "df['total'] = df[race_gender].sum(axis=1)\n",
    "print(df.isnull().sum()[lambda s: s > 0])"
   ]
  },

    "def race_share(d, level):\n",
    "    t = d[d.occupation_level == level].groupby(['submission_year','race'])['count'].sum().unstack()\n",
    "    return (t.div(t.sum(axis=1), axis=0) * 100).round(1)\n",
    "\n",
    "top_ex  = race_share(ex,   'Top management')\n",
    "top_all = race_share(long, 'Top management')\n",
    "print(top_ex.loc[[2013, 2024]]); print(top_all.loc[[2013, 2024]])\n",
    "print('Max abs. difference with vs without big employer (pp):', (top_all - top_ex).abs().max().max())\n",
    "print('Top management headcount excl. big employer:', ex[ex.occupation_level=='Top management'].groupby('submission_year')['count'].sum().loc[[2013,2024]].to_dict())\n",

    "for r in top_ex.columns:\n",
